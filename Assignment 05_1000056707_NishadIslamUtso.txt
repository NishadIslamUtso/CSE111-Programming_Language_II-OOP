//Company Class
public class Company{
  
  public Employee[] employees=new Employee[3];
  public String companyName="ABC Company";
  public int count=0;
  
  public void addEmployee(Employee e){
    
    if(count>=3){
      System.out.println("No more vacancy");
    }
    else{
      employees[count]=e;
      count++;
      System.out.println(e.name+" has joined the company");
    }
  }
  
  public void removeEmployee(Employee e){
    
    for(int i=0;i<count;i++){
      
      if(employees[i]==e){
        System.out.println(e.name+" has left the company");
        
        for(int j=i;j<count-1;j++){
          employees[j]=employees[j+1];
        }
        count--;
        break;
      }
    }
  }
  
  public void details(){
    System.out.println("Company Name: "+companyName);
    System.out.println("Total Employee: "+count);
    System.out.println("Fulltime Employees: ");
    
    for(int i=0;i<count;i++){
      
      if(employees[i].type.equals("Fulltime")){
        System.out.println("Name: "+employees[i].name+", ID: "+employees[i].id);
      }
    }
    System.out.println("Part-Time Employees: ");
    
    for(int i=0;i<count;i++){
      
      if(employees[i].type.equals("Part-time")){
        System.out.println("Name: "+employees[i].name+", ID: "+employees[i].id);
      }
    }
  }
}

//Department Class
public class Department{
  public Student[] students=new Student[5];
  public String deptName;
  public int count=0;
  
  public Department(String n){
    deptName=n;
  }
  
  public void addStudent(Student s){
    boolean exists=false;
    
    for(int j=0;j<count;j++){

      if(students[j].id==s.id){
        exists=true;
        break;
      }
    }
    
    if(exists){
      System.out.println("Student with the same ID already exists, Please try with another ID");
    }
    else{
      students[count]=s;
      count++;
      System.out.println("Welcome to "+deptName+" department, "+s.name);
    }
  }
  
  public void addStudent(Student s1,Student s2,Student s3){
    addStudent(s1);
    addStudent(s2);
    addStudent(s3);
  }
  
  public void findStudent(int id){
    boolean found=false;
    
    for(int i=0;i<count;i++){

      if(students[i].id==id){
        found=true;
        System.out.println("Student info:");
        System.out.println("Student Name: "+students[i].name);
        System.out.println("ID: "+students[i].id);
        System.out.println("CGPA:  "+students[i].cgpa);
        break;
      }
    }
    
    if(!found){
      System.out.println("Student with this ID doesn't exist, Please give a valid ID");
    }
  }
  
  public void details(){
    System.out.println("Department Name: "+deptName);
    System.out.println("Number of student:"+count);
    System.out.println("Details of the students: ");
    
    for(int i=0;i<count;i++){
      System.out.println("Student name: "+students[i].name+", ID: "+students[i].id+", cgpa: "+students[i].cgpa);
    }
  }
}

public class UberEatsUser{
  public String name;
  public String loc;
  public boolean pending;
  
  public UberEatsUser(String n, String l){
    name=n;
    loc=l;
  }
  
  public UberEatsUser(String n){
    this(n,null);
  }
  
  public void updateLocation(String l){
    if(loc!=null){
      System.out.println("We already have your location. Please place an order!");
    }
    else{
      loc=l;
      System.out.println("Update Successful!");
    }
  }
}

//Restaurant Class
public class Restaurant{
  public UberEatsUser[] users=new UberEatsUser[3];
  public String resName,resLoc;
  public int count=0;
  
  public Restaurant(String n, String l){
    resName=n;
    resLoc=l;
  }
  
  public void takeOrder(UberEatsUser x){
    if(x.pending){
      System.out.println("You already have a pending order!");
    }
    else if(x.loc==null){
      System.out.println("Location : Unknown. Please update your location information!");
    }
    else if(count>=3){
      System.out.println("We are really busy right now. Please order after some time. Thank you!");
    }
    else{
      users[count]=x;
      x.pending=true;
      count++;
      System.out.println("Your order has been added!");
    }
  }
  
  public void completeOrders(){
    if(count==0){
      System.out.println("No pending orders at the moment.");
    }
    else{
      System.out.println("Showing "+resName+"'s orders:");
      
      for(int i=0;i<count;i++){
        System.out.println("Order by "+users[i].name+" at "+users[i].loc+" completed");
        users[i].pending=false;
      }
      count=0;
    }
  }
}

public class ConnectFriends{
  public String name;
  public int maxRequests;
  public String[] sentRequests;
  public int count=0;
  
  public ConnectFriends(String n){
    this(n,2);
  }
  
  public ConnectFriends(String n, int max){
    name=n;
    maxRequests=max;
    sentRequests=new String[max];
    System.out.println("Welcome to ConnectFriends, "+name);
  }
  
  public void sendFriendRequest(ConnectFriends f){
    
    if(count>=maxRequests){
      System.out.println(name+" has reached the friend request limit!");
    }
    
    else{
      sentRequests[count]=f.name;
      count++;
      System.out.println(name+" sent a friend request to "+f.name+".");
    }
  }
  
  public void sendFriendRequest(ConnectFriends f1, ConnectFriends f2){
    sendFriendRequest(f1);
    sendFriendRequest(f2);
  }
  
  public void removeRequest(String fname){
    boolean found=false;
    
    for(int i=0;i<count;i++){
      
      if(sentRequests[i].equals(fname)){
        found=true;
        System.out.println("Reuqest to add "+fname+" is removed for "+name+".");
        
        for(int j=i;j<count-1;j++){
          sentRequests[j]=sentRequests[j+1];
        }
        count--;
        break;
      }
    }
    
    if(!found){
      System.out.println(fname+" is not in "+name+"'s sent request list.");
    }
  }
  
  public void showDetails(){
    System.out.println("User Name: "+name);
    System.out.println("Maximum number of Sent Friend Request: "+maxRequests);
    System.out.println("Total Friends Request: "+count);
    System.out.print("Sent Friends Request: ");
    
    for(int i=0;i<count;i++){
      System.out.print(sentRequests[i]+" ");
    }
    System.out.println();
  }
}
