//Book Class
public class Book extends Product {
  public String isbn;
  public String publisher;
  public Book(int i, String t ,int p ,String is ,String pub){
    super(i,t,p);
    isbn=is;
    publisher=pub;
  }
  
  public String printDetail(){
    return getIdTitlePrice()+"\nISBN: "+isbn+" Publisher: "+publisher;
  }
}
//CD Class
public class CD extends Product{
  public String band;
  public String genre;
  public int duration;
  public CD(int i, String t ,int p ,String b ,int d ,String g){
    super(i,t,p);
    band=b;
    genre=g;
    duration=d;
  }
  
  public String printDetail(){
    return getIdTitlePrice()+"\nBand: "+band+" Duration: "+duration+" minutes Genre: "+genre;
  }
}

public class ComplexNumber extends RealNumber{
  public double imag;
  public ComplexNumber(){
    super(1.0);
    imag=1.0;
  }
  public ComplexNumber(double a, double b){
    super(a);
    imag=b;
  }
  public String details(){
    return getReal()+"\nImaginaryPart: "+imag;
  }
}

public class CSEStudent extends Student{
  public static String[] course={"CSE110", "CSE111","CSE220" ,"CSE221"};
  public static int count=0;
  
  public static void details(){
    System.out.println("Total CSE Students: "+count);
    System.out.println("Available Lab Based Courses: ");
    
    for(int i= 0;i<4;i++){
      System.out.print(course[i]+" ");
    }
    System.out.println("");
  }
  public CSEStudent(String n, int i){
    super(n,i);
    count++;
  }
  
  public void addLabBasedCourse(String s){
    boolean found =false;
    
    for(int i=0;i<4;i++){
      if(course[i].equals(s)){
        found=true;
        break;
      }
    }
    if(found==true){
      courses+=s+" ";
    }
    else{
      System.out.println("It is not a lab based course!");
    }
  }
}
